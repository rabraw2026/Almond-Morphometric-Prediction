# 📊 1. Actual vs Predicted Plots
# -----------------------------------
targets = target_columns

for i, col in enumerate(targets):
    plt.figure()
    plt.scatter(y_test.iloc[:, i], y_pred[:, i])
    plt.xlabel("Actual " + col)
    plt.ylabel("Predicted " + col)
    plt.title("Actual vs Predicted - " + col)

        [y_test.iloc[:, i].min(), y_test.iloc[:, i].max()],
        [y_test.iloc[:, i].min(), y_test.iloc[:, i].max()]
    )

# -----------------------------------
# 📉 2. Residual Plots
# -----------------------------------
for i, col in enumerate(targets):
    residuals = y_test.iloc[:, i] - y_pred[:, i]
    
    plt.figure()
    plt.scatter(y_pred[:, i], residuals)
    plt.xlabel("Predicted " + col)
    plt.ylabel("Residuals")
    plt.title("Residual Plot - " + col)
    plt.axhline(y=0)
    plt.show()
